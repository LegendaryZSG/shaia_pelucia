function earPath(shape, accent) {
  switch (shape) {
    case 'long':
      return `<ellipse cx="34" cy="18" rx="9" ry="26" fill="${accent}" transform="rotate(-12 34 18)"/>
              <ellipse cx="66" cy="18" rx="9" ry="26" fill="${accent}" transform="rotate(12 66 18)"/>`;
    case 'triangle':
      return `<polygon points="26,34 34,4 46,32" fill="${accent}"/>
              <polygon points="74,34 66,4 54,32" fill="${accent}"/>`;
    case 'floppy':
      return `<ellipse cx="24" cy="42" rx="11" ry="20" fill="${accent}" transform="rotate(-18 24 42)"/>
              <ellipse cx="76" cy="42" rx="11" ry="20" fill="${accent}" transform="rotate(18 76 42)"/>`;

      return '';
  }
}

async function carregarProdutos() {
  const stateMsg = document.getElementById('state-msg');
  try {
    const res = await fetch(`${API_BASE}/api/produtos`);
    if (!res.ok) throw new Error('Resposta não OK da API');
    produtos = await res.json();
    stateMsg.style.display = 'none';
    montarFiltros();
    const filtroAtivo = document.querySelector('.pill.active')?.dataset.cat || 'todas';
    renderGrid(filtroAtivo);
  } catch (err) {
    stateMsg.textContent = 'Não foi possível carregar os produtos. A API está rodando em http://localhost:3000?';
    console.error(err);
  }
}

function montarFiltros() {
  const categorias = [...new Set(produtos.map(p => p.categoria))];
  const toolbar = document.getElementById('toolbar');
  const filtroAtivo = toolbar.querySelector('.pill.active')?.dataset.cat || 'todas';

  toolbar.innerHTML = '';

  const btnTodas = document.createElement('button');
  btnTodas.className = 'pill' + (filtroAtivo === 'todas' ? ' active' : '');
  btnTodas.textContent = 'Todas';
  btnTodas.dataset.cat = 'todas';
  btnTodas.onclick = () => selecionarFiltro('todas');
  toolbar.appendChild(btnTodas);

  categorias.forEach(cat => {
    const btn = document.createElement('button');
    btn.className = 'pill' + (filtroAtivo === cat ? ' active' : '');
    btn.textContent = cat;
    btn.dataset.cat = cat;
    btn.onclick = () => selecionarFiltro(cat);
    toolbar.appendChild(btn);
  });

  montarCategoriasCirculo(categorias, filtroAtivo);

  const footerCats = document.getElementById('footer-categorias');
  if (footerCats) {
    footerCats.innerHTML = categorias.map(c => `<li><a onclick="selecionarFiltro('${c}'); document.getElementById('grid').scrollIntoView({behavior:'smooth'});">${c}</a></li>`).join('');
  }
  const nlArt = document.getElementById('newsletter-art');
  if (nlArt) nlArt.innerHTML = plushSVG('fantasiosos');
}

function montarCategoriasCirculo(categorias, filtroAtivo) {
  const wrap = document.getElementById('categorias-circulos');
  wrap.innerHTML = '';

  // acha um produto com foto pra representar cada categoria (senão usa o desenho)
  const representante = (cat) => {
    const daCategoria = cat === 'todas' ? produtos : produtos.filter(p => p.categoria === cat);
    return daCategoria.find(p => p.imagem) || daCategoria[0] || { categoria: cat === 'todas' ? '' : cat };
  };

  const criarCirculo = (cat, label) => {
    const rep = representante(cat);
    const btn = document.createElement('button');
    btn.className = 'cat-circulo' + (filtroAtivo === cat ? ' active' : '');
    btn.onclick = () => selecionarFiltro(cat);
    btn.innerHTML = `<div class="circ">${fotoOuDesenho(rep)}</div><span>${label}</span>`;
    wrap.appendChild(btn);
  };

  criarCirculo('todas', 'Todas');
  categorias.forEach(cat => criarCirculo(cat, cat));
}

function selecionarFiltro(cat) {
  document.querySelectorAll('.pill').forEach(p => p.classList.toggle('active', p.dataset.cat === cat));
  document.querySelectorAll('.cat-circulo').forEach(c => {
    c.classList.toggle('active', c.querySelector('span').textContent === cat || (cat === 'todas' && c.querySelector('span').textContent === 'Todas'));
  });
  renderGrid(cat);
}

function statusEstoque(p) {
  if (p.estoque_atual <= 0) return { txt: 'Esgotado', cls: 'out' };
  if (p.estoque_atual <= p.estoque_minimo) return { txt: `Só ${p.estoque_atual} restando`, cls: 'low' };
  return { txt: `${p.estoque_atual} em estoque`, cls: 'ok' };
}

function renderGrid(cat) {
  const grid = document.getElementById('grid');
  grid.innerHTML = '';
  variantesPorGrupo = {};
  const lista = cat === 'todas' ? produtos : produtos.filter(p => p.categoria === cat);

  // agrupa produtos com o mesmo nome (ex: mesma pelúcia em tamanhos diferentes)
  const grupos = {};
  lista.forEach(p => {
    if (!grupos[p.nome]) grupos[p.nome] = [];
    grupos[p.nome].push(p);
  });

  Object.entries(grupos).forEach(([nome, variantes], idx) => {
    variantes.sort((a, b) => (ORDEM_TAMANHO[a.tamanho] ?? 99) - (ORDEM_TAMANHO[b.tamanho] ?? 99));
    const groupId = `g${idx}-${variantes[0].id_produto}`;
    variantesPorGrupo[groupId] = variantes;
    const primeira = variantes[0];

    const precoMin = Math.min(...variantes.map(v => Number(v.preco_venda)));
    const temEstoque = variantes.some(v => v.estoque_atual > 0);
    const rotuloPreco = variantes.length > 1
      ? `A partir de R$ ${precoMin.toFixed(2).replace('.', ',')}`
      : `R$ ${precoMin.toFixed(2).replace('.', ',')}`;

    const card = document.createElement('div');
    card.className = 'card';
    card.style.cursor = 'pointer';
    card.onclick = () => openProduct(groupId);
    card.innerHTML = `
      <button class="heart-btn" onclick="event.stopPropagation(); this.classList.toggle('fav'); this.textContent = this.classList.contains('fav') ? '❤️' : '🤍';">🤍</button>
      <div class="swatch">${fotoOuDesenho(primeira)}</div>
      <span class="cat-tag">${primeira.categoria}</span>
      <h3>${nome}</h3>
      <div class="meta">${primeira.cor || 'cor única'} · ${variantes.length > 1 ? `${variantes.length} tamanhos` : `Tamanho ${primeira.tamanho}`}</div>
      <span class="stock ${temEstoque ? 'ok' : 'out'}">${temEstoque ? 'Disponível' : 'Esgotado'}</span>
      <div class="price">${rotuloPreco}</div>
      <button class="add-btn" ${temEstoque ? '' : 'disabled'}>${variantes.length > 1 ? 'Escolher tamanho' : 'Ver produto'}</button>
    `;
    grid.appendChild(card);
  });
}

// ===== Modal de produto (escolher tamanho) =====
let produtoModalAtual = null; // { groupId, variantes, selecionado, qtd }

function openProduct(groupId) {
  const variantes = variantesPorGrupo[groupId];
  if (!variantes) return;
  const padrao = variantes.find(v => v.estoque_atual > 0) || variantes[0];
  produtoModalAtual = { groupId, variantes, selecionado: padrao, qtd: 1 };
  renderProductModal();
  document.getElementById('product-panel').classList.add('open');
}

function closeProduct() {
  document.getElementById('product-panel').classList.remove('open');
  produtoModalAtual = null;
}

function selecionarTamanhoModal(idProduto) {
  if (!produtoModalAtual) return;
  const v = produtoModalAtual.variantes.find(x => x.id_produto === idProduto);
  if (!v || v.estoque_atual <= 0) return;
  produtoModalAtual.selecionado = v;
  produtoModalAtual.qtd = 1;
  renderProductModal();
}

function mudarQtdModal(delta) {
  if (!produtoModalAtual) return;
  const max = produtoModalAtual.selecionado.estoque_atual;
  const nova = produtoModalAtual.qtd + delta;
  if (nova >= 1 && nova <= max) {
    produtoModalAtual.qtd = nova;
    document.getElementById('qtd-modal-val').textContent = nova;
  }
}

function renderProductModal() {
  const { variantes, selecionado, qtd } = produtoModalAtual;
  const primeira = variantes[0];

  document.getElementById('product-swatch').innerHTML = fotoOuDesenho(primeira);
  document.getElementById('product-cat').textContent = primeira.categoria;
  document.getElementById('product-nome').textContent = primeira.nome;
  document.getElementById('product-meta').textContent = primeira.cor || 'cor única';

  document.getElementById('size-options').innerHTML = variantes.map(v => `
    <button class="size-option ${v.id_produto === selecionado.id_produto ? 'selected' : ''}"
            ${v.estoque_atual <= 0 ? 'disabled' : ''}
            onclick="selecionarTamanhoModal(${v.id_produto})">
      ${v.tamanho}
    </button>
  `).join('');

  const st = statusEstoque(selecionado);
  const stockEl = document.getElementById('product-stock');
  stockEl.textContent = st.txt;
  stockEl.className = `stock ${st.cls}`;

  document.getElementById('product-price').textContent =
    `R$ ${Number(selecionado.preco_venda).toFixed(2).replace('.', ',')}`;

  document.getElementById('qtd-modal-val').textContent = qtd;

  const btn = document.getElementById('product-add-btn');
  btn.disabled = selecionado.estoque_atual <= 0;
  btn.textContent = selecionado.estoque_atual <= 0 ? 'Indisponível' : 'Adicionar ao carrinho';
}

function confirmarAddModal() {
  if (!produtoModalAtual) return;
  const { selecionado, qtd } = produtoModalAtual;
  addToCart(selecionado.id_produto, qtd);
  closeProduct();
  openCart();
}

function addToCart(id, quantidade = 1) {
  const produto = produtos.find(p => p.id_produto === id);
  if (!produto) return;
  if (!cart[id]) cart[id] = { produto, quantidade: 0, nomePersonalizado: '' };
  const novaQtd = cart[id].quantidade + quantidade;
  cart[id].quantidade = Math.min(novaQtd, produto.estoque_atual);
  atualizarCartUI();
}

function mudarQtd(id, delta) {
  if (!cart[id]) return;
  cart[id].quantidade += delta;
  if (cart[id].quantidade <= 0) delete cart[id];
  atualizarCartUI();
}

function atualizarCartUI() {
  const itemsEl = document.getElementById('cart-items');
  const countEl = document.getElementById('cart-count');
  const totalEl = document.getElementById('cart-total-val');
  const checkoutBtn = document.getElementById('checkout-btn');

  const entries = Object.entries(cart);
  countEl.textContent = entries.reduce((s, [, v]) => s + v.quantidade, 0);

  itemsEl.innerHTML = entries.length
    ? entries.map(([id, v]) => `
        <div class="cart-row" style="flex-direction:column;align-items:stretch;gap:6px;">
          <div style="display:flex;justify-content:space-between;align-items:center;">
            <span>${v.produto.nome}</span>
            <div class="qty-ctrl">
              <button class="qty" onclick="mudarQtd(${id}, -1)">−</button>
              <span>${v.quantidade}</span>
              <button class="qty" onclick="mudarQtd(${id}, 1)">+</button>
            </div>
          </div>
          <input type="text" placeholder="Dar um nome pra essa pelúcia (opcional)"
                 value="${v.nomePersonalizado || ''}"
                 oninput="cart[${id}].nomePersonalizado = this.value"
                 style="font-size:13px;padding:7px 10px;">
        </div>
      `).join('')
    : '<p style="color:var(--ink-soft);font-size:14px;">Carrinho vazio.</p>';

  const total = entries.reduce((s, [, v]) => s + v.quantidade * Number(v.produto.preco_venda), 0);
  totalEl.textContent = `R$ ${total.toFixed(2).replace('.', ',')}`;
  checkoutBtn.disabled = entries.length === 0;
}

async function finalizarPedido() {
  const feedback = document.getElementById('order-feedback');
  const forma_pagamento = document.getElementById('forma_pagamento').value;
  const cpf = document.getElementById('cpf_cliente').value.trim();
  const telefone = document.getElementById('telefone_cliente').value.trim();
  const rua = document.getElementById('endereco_rua').value.trim();
  const numero = document.getElementById('endereco_numero').value.trim();
  const bairro = document.getElementById('endereco_bairro').value.trim();
  const cep = document.getElementById('endereco_cep').value.trim();

  if (Object.keys(cart).length === 0) {
    feedback.style.color = '#B23A48';
    feedback.textContent = 'Adicione algum produto ao carrinho primeiro.';
    return;
  }
  if (!usuarioLogado) {
    feedback.style.color = '#B23A48';
    feedback.textContent = 'Você precisa estar logada(o) para finalizar o pedido.';
    return;
  }
  if (!cpf || !telefone || !rua || !numero || !bairro) {
    feedback.style.color = '#B23A48';
    feedback.textContent = 'Preencha CPF, telefone e o endereço completo (rua, número, bairro).';
    return;
  }

  const itens = Object.values(cart).map(v => ({
    id_produto: v.produto.id_produto,
    quantidade: v.quantidade,
    preco_unitario: v.produto.preco_venda,
    nome_personalizado: v.nomePersonalizado || null
  }));

  feedback.style.color = 'var(--ink-soft)';
  feedback.textContent = 'Enviando pedido…';

  try {
    // 1) salva CPF e telefone no cadastro do cliente
    const resCliente = await fetch(`${API_BASE}/api/clientes/${usuarioLogado.id_cliente}`, {
      method: 'PUT',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ cpf, telefone })
    });
    const dataCliente = await resCliente.json();
    if (!resCliente.ok) throw new Error(dataCliente.erro || 'Erro ao salvar CPF/telefone');

    // 2) salva o endereço vinculado ao cliente
    const resEndereco = await fetch(`${API_BASE}/api/enderecos`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ fk_id_cliente: usuarioLogado.id_cliente, rua, bairro, numero, cep: cep || null })
    });
    const dataEndereco = await resEndereco.json();
    if (!resEndereco.ok) throw new Error(dataEndereco.erro || 'Erro ao salvar endereço');

    // 3) cria o pedido
    const res = await fetch(`${API_BASE}/api/pedidos`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ id_cliente: usuarioLogado.id_cliente, forma_pagamento, itens })
    });
    const data = await res.json();
    if (!res.ok) throw new Error(data.erro || 'Erro ao criar pedido');

    feedback.style.color = 'var(--sage-deep)';
    feedback.textContent = `Pedido #${data.id_pedido} criado com sucesso!`;
    cart = {};
    atualizarCartUI();
    carregarProdutos();
  } catch (err) {
    feedback.style.color = '#B23A48';
    feedback.textContent = err.message;
  }
}

let usuarioLogado = null; // { id_cliente, nome, email }
let abaAtual = 'entrar';

function mudarAba(aba) {
  abaAtual = aba;
  document.getElementById('tab-entrar').classList.toggle('active', aba === 'entrar');
  document.getElementById('tab-criar').classList.toggle('active', aba === 'criar');
  document.getElementById('campo-nome').style.display = aba === 'criar' ? 'block' : 'none';
  document.getElementById('login-title').textContent = aba === 'criar' ? 'Criar conta' : 'Entrar';
  document.getElementById('login-submit').textContent = aba === 'criar' ? 'Criar conta' : 'Entrar';
  document.getElementById('login-feedback').textContent = '';
}

async function entrar() {
  const feedback = document.getElementById('login-feedback');
  const email = document.getElementById('login_email').value.trim();
  const senha = document.getElementById('login_senha').value;
  const nome = document.getElementById('cad_nome').value.trim();

  if (!email || !senha) {
    feedback.style.color = '#B23A48';
    feedback.textContent = 'Preencha e-mail e senha.';
    return;
  }
  if (abaAtual === 'criar' && !nome) {
    feedback.style.color = '#B23A48';
    feedback.textContent = 'Preencha seu nome.';
    return;
  }

  feedback.style.color = 'var(--ink-soft)';
  feedback.textContent = abaAtual === 'criar' ? 'Criando conta…' : 'Entrando…';

  try {
    const rota = abaAtual === 'criar' ? '/api/cadastro' : '/api/login';
    const corpo = abaAtual === 'criar' ? { nome, email, senha } : { email, senha };

    const res = await fetch(`${API_BASE}${rota}`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(corpo)
    });
    const data = await res.json();
    if (!res.ok) throw new Error(data.erro || 'Não foi possível entrar.');

    usuarioLogado = data;
    localStorage.setItem('shaia_usuario', JSON.stringify(data));
    mostrarApp();
  } catch (err) {
    feedback.style.color = '#B23A48';
    feedback.textContent = err.message;
  }
}

function sair() {
  usuarioLogado = null;
  localStorage.removeItem('shaia_usuario');
  document.getElementById('loja-conteudo').style.display = 'none';
  document.getElementById('login-screen').style.display = 'flex';
  document.getElementById('login_email').value = '';
  document.getElementById('login_senha').value = '';
}

function mostrarApp() {
  document.getElementById('login-screen').style.display = 'none';
  document.getElementById('loja-conteudo').style.display = 'block';
  document.getElementById('app-content').style.display = 'block';
  const avatar = document.getElementById('header-avatar');
  avatar.textContent = (usuarioLogado.nome || '?').trim().charAt(0).toUpperCase();
  document.getElementById('hero-art').innerHTML = plushSVG('urso');
  carregarProdutos();
}

// Se já tinha uma sessão salva no navegador, entra direto
const sessaoSalva = localStorage.getItem('shaia_usuario');
if (sessaoSalva) {
  usuarioLogado = JSON.parse(sessaoSalva);
  mostrarApp();

}
